% 式（1）
\[
\begin{gathered}
\bar r=\frac{1+2+\cdots+N}{N}=\frac{N+1}{2} \\

\end{gathered}
\]

% 式（2）
\[
\begin{gathered}
B=\sum_{i=1}^{k}n_i(\bar r_i-\bar r)^2 \\
B=6(16.25-12.5)^2+\cdots \\
\quad +6(4.3333\ldots-12.5)^2 \\
B=546.5833\ldots
\end{gathered}
\]

% 式（3）
\[
\begin{gathered}
s_r^2=\frac{1}{N-1}\sum_{j=1}^{N}(r_j-\bar r)^2 \\
\operatorname{E}_0(Y_a)=\bar r \\
\operatorname{Var}_0(Y_a)=\frac{N-1}{N}s_r^2
\end{gathered}
\]

% 式（3a）
\[
\begin{gathered}
0=\operatorname{Var}_0\!\left(\sum_{a=1}^{N}Y_a\right) \\
\quad=N\operatorname{Var}_0(Y_1) \\
\qquad+N(N-1)\operatorname{Cov}_0(Y_1,Y_2) \\
\operatorname{Cov}_0(Y_a,Y_b)=-\frac{s_r^2}{N} \\
\qquad (a\ne b)
\end{gathered}
\]

% 式（3b）
\[
\begin{gathered}
\operatorname{Var}_0(\bar Y_i)=\frac{1}{n_i^2}\left[n_i\frac{N-1}{N}s_r^2\right. \\
\qquad\left.-n_i(n_i-1)\frac{s_r^2}{N}\right] \\
\quad=s_r^2\left(\frac{1}{n_i}-\frac{1}{N}\right) \\
\operatorname{Cov}_0(\bar Y_i,\bar Y_\ell)=-\frac{s_r^2}{N} \\
\qquad (i\ne\ell)
\end{gathered}
\]

% 式（3c）
\[
\begin{gathered}
\operatorname{E}_0(B)=\sum_i n_i\operatorname{Var}_0(\bar Y_i) \\
\quad=s_r^2\sum_i\left(1-\frac{n_i}{N}\right) \\
\quad=(k-1)s_r^2 \\
H=\frac{B}{s_r^2},\qquad \operatorname{E}_0(H)=k-1
\end{gathered}
\]

% 式（4）
\[
\begin{gathered}
\sum_{j=1}^{N}(j-\bar r)^2=\frac{N(N^2-1)}{12} \\
s_{\mathrm{base}}^2=\frac{N(N+1)}{12} \\
H^*=\frac{B}{s_{\mathrm{base}}^2}=\frac{12B}{N(N+1)}
\end{gathered}
\]

% 式（5）
\[
\begin{gathered}
R_i=n_i\bar r_i,\qquad \sum_iR_i=N\bar r \\
B=\sum_i\frac{R_i^2}{n_i}-N\bar r^2 \\
H^*=\frac{12}{N(N+1)}\sum_i\frac{R_i^2}{n_i} \\
\qquad -3(N+1)
\end{gathered}
\]
